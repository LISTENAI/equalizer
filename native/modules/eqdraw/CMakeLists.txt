add_library(lsaudio_eqdraw SHARED src/eqdraw.c)
target_link_libraries(lsaudio_eqdraw PUBLIC lsaudio_eqdrc)
target_include_directories(lsaudio_eqdraw PUBLIC include)
target_compile_definitions(lsaudio_eqdraw PRIVATE LSAUDIO_EQDRAW_BUILD)
set_target_properties(lsaudio_eqdraw PROPERTIES C_VISIBILITY_PRESET hidden)
if(APPLE)
  set_target_properties(lsaudio_eqdraw PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(lsaudio_eqdraw PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
if(MSVC)
  target_compile_options(lsaudio_eqdraw PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(lsaudio_eqdraw PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()

if(BUILD_TESTING)
  add_executable(eqdraw_wrapper_test ../../tests/eqdraw/wrapper_test.c)
  target_link_libraries(eqdraw_wrapper_test PRIVATE lsaudio_eqdraw)
  add_test(NAME eqdraw_wrapper COMMAND eqdraw_wrapper_test)
endif()
