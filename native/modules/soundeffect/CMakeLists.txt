add_library(lsaudio_soundeffect SHARED src/soundeffect.c ../../common/biquad.c)
target_include_directories(lsaudio_soundeffect PUBLIC include ../eqdrc/include PRIVATE ../../common)
target_compile_definitions(lsaudio_soundeffect PRIVATE LSAUDIO_SOUNDEFFECT_BUILD)
set_target_properties(lsaudio_soundeffect PROPERTIES C_VISIBILITY_PRESET hidden)
if(MSVC)
  target_compile_options(lsaudio_soundeffect PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(lsaudio_soundeffect PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  target_link_libraries(lsaudio_soundeffect PRIVATE m)
endif()

  add_executable(soundeffect_test ../../tests/soundeffect/sound_test.c)
  target_link_libraries(soundeffect_test PRIVATE lsaudio_soundeffect)
  add_test(NAME soundeffect COMMAND soundeffect_test)
endif()
